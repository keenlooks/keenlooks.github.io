<script>
  (function () {
    /* "Search this site" opens the command palette (assets/js/palette.js)
       prefilled with the missing page's last path segment, turned into
       words. Block comments only: production HTML is compressed. */
    var btn = document.getElementById('gol404-search');
    if (!btn) return;
    function guess() {
      var parts = window.location.pathname.split('/');
      var last = '';
      for (var i = parts.length - 1; i >= 0 && !last; i--) last = parts[i];
      try { last = decodeURIComponent(last); } catch (e) {}
      last = last.replace(/\.html?$/i, '').replace(/[-_.+]+/g, ' ').replace(/\s+/g, ' ').trim();
      if (/^404$/.test(last)) last = '';
      return last.slice(0, 60);
    }
    btn.hidden = false;
    btn.addEventListener('click', function () {
      if (window.__cmdk && window.__cmdk.open) window.__cmdk.open(guess());
    });
  })();
</script>
